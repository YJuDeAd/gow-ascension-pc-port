### Task 1: Update SPU Workload Registry (`spu_workloads.c`)
Add image 5 (`spu_wws_job_pm`) declarations and registration to `./src/gen/spu/spu_workloads.c` so `spu_workload_find_img` resolves fingerprint `0xC8D774963A96DCF2ULL` to entry point `spu_wws_job_pm_spu_func_00000000`.

```powershell
python -c "
path = 'src/gen/spu/spu_workloads.c'
content = open(path, 'r', encoding='utf-8').read()

decl = '''extern void spu_wws_job_pm_spu_func_00000000(spu_context*);
extern void spu_wws_job_pm_spu_recomp_register(void);
'''

reg = '''    spu_begin_image(5); spu_wws_job_pm_spu_recomp_register();
    spu_workload_register_img(0xC8D774963A96DCF2ULL, spu_wws_job_pm_spu_func_00000000, 5, \"wws_job_pm\");
    spu_workload_register_img(0x5B74C347FC8A0D40ULL, spu_wws_job_pm_spu_func_00000000, 5, \"wws_job_pm.elf\");
    spu_begin_image(0);
'''

# Insert decl before void gow_spu_register_all
content = content.replace('void gow_spu_register_all(void)', decl + '\nvoid gow_spu_register_all(void)')

# Replace spu_begin_image(0); with new registration
content = content.replace('    spu_begin_image(0);', reg)

open(path, 'w', encoding='utf-8', newline='\n').write(content)
print('Updated src/gen/spu/spu_workloads.c with image 5 registration.')
"
```
